const SCRYPT_N=16384, SCRYPT_R=8, SCRYPT_P=1, KEYLEN=32;
function hashPassword(password){
  const salt=crypto.randomBytes(16);
  const derived=crypto.scryptSync(String(password),salt,KEYLEN,{N:SCRYPT_N,r:SCRYPT_R,p:SCRYPT_P,maxmem:64*1024*1024});
  return `scrypt$${SCRYPT_N}$${SCRYPT_R}$${SCRYPT_P}$${salt.toString('base64url')}$${derived.toString('base64url')}`;
}
function safeEqual(a,b){const aa=Buffer.from(String(a));const bb=Buffer.from(String(b));return aa.length===bb.length&&crypto.timingSafeEqual(aa,bb)}
function verifyPassword(password,stored){
  if(!stored)return false;
  const parts=String(stored).split('$');
  if(parts[0]==='scrypt'&&parts.length===6){
    try{const [,n,r,p,salt,hash]=parts;const derived=crypto.scryptSync(String(password),Buffer.from(salt,'base64url'),KEYLEN,{N:Number(n),r:Number(r),p:Number(p),maxmem:64*1024*1024});return safeEqual(derived.toString('base64url'),hash)}catch{return false}
  }
  // Backward-compatible verification for v5/v6 records. Successful legacy logins are rehashed below.
  try{return safeEqual(crypto.scryptSync(String(password),'codesazan-salt',32).toString('hex'),String(stored))}catch{return false}
}

function verify(token){
  try{
    const [body,mac]=String(token||'').split('.');
    if(!body||!mac||!crypto.timingSafeEqual(Buffer.from(mac),Buffer.from(sign(body))))return null;
    const payload=JSON.parse(Buffer.from(body,'base64url').toString('utf8'));
    if(!payload.exp||payload.exp<Math.floor(Date.now()/1000))return null;
    return payload;
  }catch{return null}
}

async function serveHls(req,res,next){
  try{
    const payload=access.validateMediaToken(req.query.token,req.params.mediaId);
    if(!payload)return res.status(401).end();
    const mediaRoot=path.resolve(process.env.MEDIA_DIR||'/app/media');
    const base=path.resolve(mediaRoot,'hls',req.params.mediaId);
    const requested=req.params.quality==='master'?'master.m3u8':path.join(req.params.quality,req.params.file);
    const target=path.resolve(base,requested);
    if(!target.startsWith(base+path.sep)||!fs.existsSync(target))return res.status(404).end();
    if(!target.toLowerCase().endsWith('.m3u8')){
      res.set('Cache-Control','private, max-age=3600');
      return res.sendFile(target);
    }
    let body=fs.readFileSync(target,'utf8');
    const signed=encodeURIComponent(req.query.token);
    body=body.split(/\r?\n/).map(line=>{
      const trimmed=line.trim();
      if(!trimmed||trimmed.startsWith('#'))return line;
      if(/\.(m3u8|ts|m4s|mp4)$/i.test(trimmed))return `${trimmed}?token=${signed}`;
      return line;
    }).join('\n');
    res.set({'Content-Type':'application/vnd.apple.mpegurl','Cache-Control':'private, no-store'}).send(body);
  }catch(e){next(e)}
}

function deepMerge(base, extra){if(!extra||typeof extra!=='object')return base;const out={...base};for(const [k,v] of Object.entries(extra)){if(v&&typeof v==='object'&&!Array.isArray(v)&&base[k]&&typeof base[k]==='object'&&!Array.isArray(base[k]))out[k]=deepMerge(base[k],v);else out[k]=v;}return out;}
